== Add details

[source, java]
----
@Operation(
  summary = "Create a new pet",
  description = "Create a new pet after validating it is not already exist in the database",
  operationId = "createPet",
  tags = { "pet-controller-v-1" },
  responses = {
    @ApiResponse(responseCode = "201", description = "Pet created", content = {
      @Content(mediaType = "application/json", schema = @Schema(implementation = PetDto.class))
    }),
    @ApiResponse(responseCode = "400", description = "The request is invalid", content = {
      @Content(mediaType = "application/json", schema = @Schema(implementation = MyError.class))
    })
  }
)
@PostMapping("/pets")
public ResponseEntity<PetDto> createPet(@RequestBody PetDto pet) {
  var domain = toDomain(pet);
  var result = service.createPet(domain);
  return ResponseEntity.status(HttpStatus.ACCEPTED).body(toDto(result));
}
----

=== Plugin tweak

[source, xml, numbered]
----
<plugin>
  <groupId>org.openapitools</groupId>
  <artifactId>openapi-generator-maven-plugin</artifactId>
  <executions>
    <execution>
      <id>server</id>
      <goals>
          <goal>generate</goal>
      </goals>
      <configuration>
        <inputSpec>${project.basedir}/src/main/resources/api.yml</inputSpec>
        <generatorName>spring</generatorName>
        <apiPackage>bzh.zomzog.prez.contract_first.demo.controller.api</apiPackage>
        <modelPackage>bzh.zomzog.prez.contract_first.demo.controller.dto</modelPackage>
        <generateSupportingFiles>true</generateSupportingFiles>
        <modelNameSuffix>Dto</modelNameSuffix>
        <schemaMappings>
          <schemaMapping>Pageable=org.springframework.data.domain.Pageable</schemaMapping>
        </schemaMappings>
        <configOptions>
          <interfaceOnly>true</interfaceOnly>
          <useSpringBoot3>true</useSpringBoot3>
          <useTags>true</useTags>
          <delegatePattern>true</delegatePattern>
          <useSwaggerUI>false</useSwaggerUI>
        </configOptions>
      </configuration>
    </execution>
  </executions>
</plugin>
----

[NOTE.speaker]

[source,yaml]
----
paths:
  /ponies/{ponyName}:

    get:
      summary: "Get one"

  parameters:
    NameInPath:
      name: ponyName
      in: path
      schema:
        $ref: "#/components/schemas/Name"
      required: true
